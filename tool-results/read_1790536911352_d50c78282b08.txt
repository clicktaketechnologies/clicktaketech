   374→// ============================ LOGIN ============================
   375→function LoginGate({ onLogin }: { onLogin: (t: string, perms: Set<string> | null) => void }) {
   376→  const { toast } = useToast();
   377→  const [email, setEmail] = useState("admin@clicktaketech.com");
   378→  const [password, setPassword] = useState("");
   379→  const [loading, setLoading] = useState(false);
   380→
   381→  const submit = async (e: React.FormEvent) => {
   382→    e.preventDefault();
   383→    setLoading(true);
   384→    try {
   385→      const res = await fetch("/api/admin/auth", {
   386→        method: "POST",
   387→        headers: { "Content-Type": "application/json" },
   388→        body: JSON.stringify({ email, password }),
   389→      });
   390→      const data = await res.json().catch(() => ({}));
   391→      if (!res.ok || !data.ok) throw new Error(data.error || "Login failed");
   392→      const perms = parsePermissions(data.user?.permissions ?? null);
   393→      try {
   394→        localStorage.setItem(TOKEN_KEY, data.token);
   395→        localStorage.setItem("clicktake_admin_perms", data.user?.permissions ?? "null");
   396→      } catch {
   397→        /* ignore */
   398→      }
   399→      onLogin(data.token, perms);
   400→      toast({ title: "Logged in", description: "Welcome to the admin panel." });
   401→    } catch (err) {
   402→      toast({
   403→        title: "Login failed",
   404→        description: err instanceof Error ? err.message : "Please try again.",
   405→        variant: "destructive",
   406→      });
   407→    } finally {
   408→      setLoading(false);
   409→    }
   410→  };
   411→
   412→  return (
   413→    <div className="flex min-h-screen items-center justify-center bg-background px-4">
   414→      <form
   415→        onSubmit={submit}
   416→        className="w-full max-w-md rounded-3xl border border-border/50 bg-card/40 p-8 ring-gradient"
   417→      >
   418→        <div className="flex items-center gap-3">
   419→          <span className="flex h-12 w-12 items-center justify-center rounded-xl bg-brand-gradient text-white">
   420→            <ShieldCheck className="h-6 w-6" />
   421→          </span>
   422→          <div>
   423→            <h1 className="text-xl font-bold tracking-tight">Admin Login</h1>
   424→            <p className="text-xs text-muted-foreground">ClickTake Technologies CMS</p>
   425→          </div>
   426→        </div>
   427→        <div className="mt-6 space-y-4">
   428→          <div>
   429→            <Label className="text-sm">Email</Label>
   430→            <Input
   431→              type="email"
   432→              value={email}
   433→              onChange={(e) => setEmail(e.target.value)}
   434→              className="mt-1.5 bg-background/50"
   435→              required
   436→            />
   437→          </div>
   438→          <div>
   439→            <Label className="text-sm">Password</Label>
   440→            <Input
   441→              type="password"
   442→              value={password}
   443→              onChange={(e) => setPassword(e.target.value)}
   444→              placeholder="••••••••"
   445→              className="mt-1.5 bg-background/50"
   446→              required
   447→            />
   448→          </div>
   449→          <Button
   450→            type="submit"
   451→            disabled={loading}
   452→            className="w-full bg-brand-gradient text-white"
   453→          >
   454→            {loading ? <Loader2 className="mr-2 h-4 w-4 animate-spin" /> : null}
   455→            {loading ? "Signing in…" : "Sign in"}
   456→          </Button>
   457→          <p className="text-center text-xs text-muted-foreground">
   458→            Demo: admin@clicktaketech.com · clicktake-admin-2026
   459→          </p>
   460→        </div>
   461→      </form>
   462→    </div>
   463→  );
   464→}

   466→// ============================ HOOK ============================
   467→function useAdminFetch(token: string) {
   468→  return useCallback(
   469→    async (path: string, opts?: RequestInit) => {
   470→      const res = await fetch(path, {
   471→        ...opts,
   472→        headers: {
   473→          ...(opts?.body ? { "Content-Type": "application/json" } : {}),
   474→          "x-admin-token": token,
   475→          ...(opts?.headers || {}),
   476→        },
   477→      });
   478→      return res;
   479→    },
   480→    [token]
   481→  );
   482→}
